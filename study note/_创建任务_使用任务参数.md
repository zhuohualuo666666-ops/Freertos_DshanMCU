int LCD_PrintSignedVal(uint8_t x, uint8_t y, int32_t val)
{
     char str[16];
    char revert_str[16];
    int i = 0, j = 0;
    int mod;

    if (val < 0)
    {
        str[i++] = '-';
        val = 0 - val;
    }
    else if (val == 0)
    {
        str[i++] = '0';
    }

    while (val)
    {
        mod = val % 10;
        revert_str[j++] = mod + '0';
        val = val / 10;
    }

    while (j)
    {
        str[i++] = revert_str[j-1];
        j--;
    } 
    
    str[i] = '\0';
    OLED_PrintString(x, y, str);
    return i;
}

//屏幕打印符号函数

int LCD_PrintString(uint8_t x, uint8_t y, const char *str)
{
   int i = 0;
    while (str[i])
    {
        OLED_PutChar(x, y, str[i]);
        x++;
        if(x > 15)
        {
            x  = 0;
            y += 2;
        }
                
        i++;
    }
    return i;
}
//屏幕打印字符串函数

struct TaskPrintInfo {
	uint8_t x;
	uint8_t y;
	char name[16];
};
//结构体里保存LCD屏幕的横纵坐标及打印信息

static struct TaskPrintInfo g_Task1Info = {0, 0, "Task1"};
static struct TaskPrintInfo g_Task2Info = {0, 3, "Task2"};
static struct TaskPrintInfo g_Task3Info = {0, 6, "Task3"};
//三个结构体变量对应三个任务,结构体参数分别为横坐标，纵坐标，任务名

static int g_LCDCanUse = 1;
//使用一个全局变量来保护打印过程（有缺陷）

void LcdPrintTask(void *params)//被三个任务调用的函数
{
	struct TaskPrintInfo *pInfo = params;//结构体指针传递参数
	uint32_t cnt = 0;
	int len;
	
	while (1)
	{
		/* 打印信息 */
		if (g_LCDCanUse)
		{
			g_LCDCanUse = 0;//置为0表示还没打印完
			len = LCD_PrintString(pInfo->x, pInfo->y, pInfo->name);//打印任务名称
			len += LCD_PrintString(len, pInfo->y, ":");//打印冒号
			LCD_PrintSignedVal(len, pInfo->y, cnt++);//
			g_LCDCanUse = 1;//置为1表示打印完成
		}
		mdelay(500);//延时函数目的是延长任务运行时间，使调度器调度发生在这里的概率更高，达到保护打印过程的目的
	}
}


void MX_FREERTOS_Init(void) 
{
  
	TaskHandle_t xSoundTaskHandle;
	BaseType_t ret;
	
	LCD_Init();
	LCD_Clear();

  extern void PlayMusic(void *params);

  /* 使用同一个函数创建不同的任务 */
  xTaskCreate(LcdPrintTask, "task1", 128, &g_Task1Info, osPriorityNormal, NULL);
  xTaskCreate(LcdPrintTask, "task2", 128, &g_Task2Info, osPriorityNormal, NULL);

}
